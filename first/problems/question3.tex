\begin{question}{Efficiently Simulating Bounded Stacks with RNNs}

\begin{subquestion}
    The language $D(2, 3)$ consists of brackets of $k = 2$ types with maximum nesting depth $m = 3$. 
    We encode the unclosed opening brackets using a stack whose \emph{rightmost} symbol is the top of the stack.
    $[\openBr{1}\openBr{2}]$ transitions to $[\openBr{1}]$ after reading $\closeBr{2}$.
    \begin{itemize}
        \item Reading a closing bracket $\closeBr{i}$: pop the top of the stack if it equals $\openBr{i}$; otherwise (mismatch or empty stack), reject.
        \item Reading an opening bracket $\openBr{i}$: push it on top, unless the stack already has $m = 3$ symbols, in which case the depth bound is exceeded and we reject.
    \end{itemize}

\begin{subquestion}
    We construct a DFA $\mathcal{A} = \fsatuple$ that recognizes $D(k, m)$ by encoding every legal stack configuration as a state. 
    Let $\openBr{} = \{\openBr{1}, \dots, \openBr{k}\}$ and $\closeBr{} = \{\closeBr{1}, \dots, \closeBr{k}\}$.
    
    \textbf{Alphabet:} $\alphabet = \openBr{} \cup \closeBr{}$, so $\nsymbols = 2k$.
    
    \textbf{States:} $\states = \{[\stackseq] \mid \stackseq \in \openBr{}^{\le m}\} \cup \{\rejectState\}$, where $[\stackseq]$ corresponds to the stack of unclosed 
    brackets $\stackseq$ with $|\stackseq| \le m$, and $\rejectState$ is an absorbing rejecting state. $\acceptState \defeq [\emptystack]$ for the empty-stack state.
    
    \textbf{Initial states:} $\initial = \{\acceptState\}$ (start with an empty stack).
    
    \textbf{Final states:} $\final = \{\acceptState\}$ (only the empty stack accepts).
    
    \textbf{Transition $\trans : \states \times \alphabet \to \states$:} For valid prefix $\stackseq \in \openBr{}^{*}$ with $[\stackseq] \in \states$ and $i, j \in \{1, \dots, k\}$:
    
    \begin{enumerate}
        \item \emph{Valid push} (if $|\stackseq| < m$): $\trans([\stackseq], \openBr{i}) = [\stackseq\, \openBr{i}]$.
        \item \emph{Valid pop}: $\trans([\stackseq\, \openBr{i}], \closeBr{i}) = [\stackseq]$.
        \item \emph{Depth overflow} (if $|\stackseq| = m$): $\trans([\stackseq], \openBr{i}) = \rejectState$.
        \item \emph{Mismatched close} (if $i \ne j$): $\trans([\stackseq\, \openBr{j}], \closeBr{i}) = \rejectState$.
        \item \emph{Underflow}: $\trans(\acceptState, \closeBr{i}) = \rejectState$.
        \item \emph{Sink}: $\trans(\rejectState, s) = \rejectState$ for every $s \in \alphabet$.
    \end{enumerate}
    These cases specify $\trans$ on every $(\states \setminus \{\rejectState\}) \times \alphabet$ pair as well as on $\{\rejectState\} \times \alphabet$, so $\trans$ is deterministic.

    \paragraph{Correctness.} A string $\str$ is in $D(k, m)$ iff (i) the brackets are balanced and (ii) the running depth $d(\str_{\le t}) \le m$.
    The state $[\stackseq_t]$ reached after reading $\str_{\le t}$ stores the multiset \emph{and order} of currently unclosed brackets, so $|\stackseq_t| = d(\str_{\le t})$. 
    The transitions enforce the two conditions:
    \begin{itemize}
        \item Any depth violation triggers rule (3), sending $\mathcal{A}$ into the absorbing $\rejectState$.
        \item Any mismatched close (rule 4) or underflow (rule 5) likewise sends $\mathcal{A}$ to $\rejectState$.
    \end{itemize}
    Conversely, while $\str_{\le t} \in D(k, m)$ at every prefix, only rules (1) and (2) fire and the automaton stays inside $\{[\stackseq] \mid \stackseq \in \openBr{}^{\le m}\}$. 
    Since $\final = \{\acceptState\}$, the run accepts iff after reading the whole string the stack is empty, i.e.\ all opens were correctly matched. Hence $\mathcal{L}(\mathcal{A}) = D(k, m)$.

    \paragraph{Complexity.}
    \[
        |\states| \;=\; 1 + \sum_{j=0}^{m} k^{j} \;=\; 1 + \frac{k^{m+1} - 1}{k - 1} \;=\; \mathcal{O}(k^{m}).
    \]
    Because $\mathcal{A}$ is a DFA, the number of transitions is
    \[
        |\trans| \;=\; |\states| \cdot |\alphabet| \;=\; \mathcal{O}(k^{m}) \cdot 2k \;=\; \mathcal{O}(k^{m+1}).
    \]
\end{subquestion}

\begin{subquestion}
    Let $\mathbf{I}_k$ and $\mathbf{0}_k$ denote the $k\times k$ identity and zero matrices, and write $\mathbf{h}$ as $m$ stacked $k$-blocks where block~$j$ is the $j$-th symbol from the top of the stack. 
    Block~1 is the top, blocks beyond the stack depth are zero.
    Pushing onto the stack means \emph{shifting all $k$-blocks one position down} and writing the new symbol into block~1. 
    Popping means \emph{shifting all $k$-blocks one position up} and zeroing block~$m$. 
    Realize this with two block-banded $km \times km$ matrices,
    \[
        \mathbf{U}_{\text{push}} \;=\;
        \begin{pmatrix}
            \mathbf{0}_k & \mathbf{0}_k & \cdots & \mathbf{0}_k & \mathbf{0}_k \\
            \mathbf{I}_k & \mathbf{0}_k & \cdots & \mathbf{0}_k & \mathbf{0}_k \\
            \mathbf{0}_k & \mathbf{I}_k & \cdots & \mathbf{0}_k & \mathbf{0}_k \\
            \vdots & \vdots & \ddots & \vdots & \vdots \\
            \mathbf{0}_k & \mathbf{0}_k & \cdots & \mathbf{I}_k & \mathbf{0}_k
        \end{pmatrix},
        \qquad
        \mathbf{U}_{\text{pop}} \;=\;
        \begin{pmatrix}
            \mathbf{0}_k & \mathbf{I}_k & \mathbf{0}_k & \cdots & \mathbf{0}_k \\
            \mathbf{0}_k & \mathbf{0}_k & \mathbf{I}_k & \cdots & \mathbf{0}_k \\
            \vdots & \vdots & \vdots & \ddots & \vdots \\
            \mathbf{0}_k & \mathbf{0}_k & \mathbf{0}_k & \cdots & \mathbf{I}_k \\
            \mathbf{0}_k & \mathbf{0}_k & \mathbf{0}_k & \cdots & \mathbf{0}_k
        \end{pmatrix},
    \]
    and define the recurrence-matrix function $\mathbf{U} : \overline{\alphabet} \to \mathbb{R}^{km \times km}$ by
    \[
        \mathbf{U}(y_t) \;=\;
        \begin{cases}
            \mathbf{U}_{\text{push}} & \text{if } y_t \in \{\openBr{1}, \dots, \openBr{k}\}, \\
            \mathbf{U}_{\text{pop}}  & \text{if } y_t \in \{\closeBr{1}, \dots, \closeBr{k}\}, \\
            \mathbf{I}_{km}          & \text{if } y_t = \eos.
        \end{cases}
    \]
    The choice $\mathbf{U}(\eos) = \mathbf{I}_{km}$ keeps the stack representation intact at termination, since $\eos$ is not a stack operation. 
    The input matrix $\mathbf{V} \in \mathbb{R}^{km \times (2k+1)}$ injects the new opening bracket into block~1 only when $y_t$ is an opening bracket, so we set
    \[
        \mathbf{V} \;=\;
        \begin{pmatrix}
            \mathbf{I}_k \\
            \mathbf{0}_k \\
            \vdots \\
            \mathbf{0}_k
        \end{pmatrix}
        \begin{pmatrix} \mathbf{I}_k & \mathbf{0}_k & \mathbf{0}_{k\times 1} \end{pmatrix}
        \;=\;
        \begin{pmatrix}
            \mathbf{I}_k & \mathbf{0}_k & \mathbf{0}_{k\times 1} \\
            \mathbf{0}_k & \mathbf{0}_k & \mathbf{0}_{k\times 1} \\
            \vdots & \vdots & \vdots \\
            \mathbf{0}_k & \mathbf{0}_k & \mathbf{0}_{k\times 1}
        \end{pmatrix},
    \]
    where the columns are ordered as $\openBr{1}, \dots, \openBr{k}, \closeBr{1}, \dots, \closeBr{k}, \eos$. 
    With this $\mathbf{V}$, $\mathbf{V}\mathbf{o}_{\openBr{i}} = (\mathbf{e}_i, \mathbf{0}_{(m-1)k})^\top$ and $\mathbf{V}\mathbf{o}_{\closeBr{i}} = \mathbf{V}\mathbf{o}_{\eos} = \mathbf{0}_{km}$.

    \paragraph{Correspondence with $\mathcal{A}$.} Suppose $\mathbf{h}_{t-1}$ encodes a non-rejecting state $q = [\openBr{i_1}\cdots\openBr{i_{m'}}] \in \states$ via Eq.~(27). We check the three transition kinds.

    \emph{Valid push.} If $y_t = \openBr{i}$ and $m' < m$, $\mathcal{A}$ moves to $q' = [\openBr{i_1}\cdots\openBr{i_{m'}}\openBr{i}]$. 
    Multiplying out, $\mathbf{U}_{\text{push}}\mathbf{h}_{t-1}$ takes block~$j$ of $\mathbf{h}_{t-1}$ into block~$j{+}1$ for $j = 1, \dots, m{-}1$ and zeros block~1. 
    Adding $\mathbf{V}\mathbf{o}_{\openBr{i}}$ writes $\mathbf{e}_i$ into block~1. 
    The resulting blocks are $(\mathbf{e}_i, \mathbf{e}_{i_{m'}}, \dots, \mathbf{e}_{i_1}, \mathbf{0}, \dots, \mathbf{0})$, exactly $h(q')$.

    \emph{Valid pop.} If $y_t = \closeBr{i}$ with $i = i_{m'}$, $\mathcal{A}$ moves to $q' = [\openBr{i_1}\cdots\openBr{i_{m'-1}}]$. 
    Now $\mathbf{U}_{\text{pop}}\mathbf{h}_{t-1}$ takes block~$j$ into block~$j{-}1$ for $j = 2, \dots, m$ and zeros block~$m$, while $\mathbf{V}\mathbf{o}_{\closeBr{i}} = \mathbf{0}$. 
    The resulting blocks are $(\mathbf{e}_{i_{m'-1}}, \dots, \mathbf{e}_{i_1}, \mathbf{0}, \dots, \mathbf{0})$, exactly $h(q')$.

    \emph{$\eos$.} The DFA does not consume $\eos$ (cf.\ part (b)).
    We simply preserve the stack via $\mathbf{U}(\eos) = \mathbf{I}_{km}$ and $\mathbf{V}\mathbf{o}_{\eos} = \mathbf{0}$, so $\mathbf{h}_t = \mathbf{h}_{t-1}$.

    \emph{Invalid transitions.} Pushing on a full stack ($m' = m$) loses block~$m$ (shifted out), and popping a mismatched or empty stack discards the wrong $\mathbf{e}$ or stays at $\mathbf{0}$. 
    In either case $\mathbf{h}_t$ no longer encodes valid stack. The dynamics map alone cannot detect this. 
    It is the recognition module of part~(f) that assigns probability~$<\eta$ to such continuations, so in the locally $\eta$-truncated support these strings are rejected.

    Hence, on every \emph{valid} prefix, the RNN's hidden states are in one-to-one correspondence with the states $\mathcal{A}$ visits, and the recurrence $\mathbf{h}_t = \mathbf{U}(y_t)\mathbf{h}_{t-1} + \mathbf{V}\mathbf{o}_{y_t}$ implements $\trans$.
\end{subquestion}

\begin{subquestion}
    Since $\mathbf{U}$ is now fixed, it can't depend on whether $y_t$ should push or pop. 
    Following the hint, double the hidden state to $\mathbb{R}^{2km}$ and perform \emph{both} operations simultaneously in each half:
    \[
        \mathbf{h}_t \;=\; \begin{pmatrix} \mathbf{h}_t^{(\text{p})} \\ \mathbf{h}_t^{(\text{q})} \end{pmatrix} \in \mathbb{R}^{2km},
    \]
    where the top half $\mathbf{h}_t^{(\text{p})}$ holds the result of a push and the bottom half $\mathbf{h}_t^{(\text{q})}$ holds the result of a pop. 
    The Heaviside activation $H$ together with large negative entries in $\mathbf{V}$ then gates off whichever half does not match the input. 
    We maintain the invariant that one half is non-zero and equals the correct stack encoding $h([\stackseq])$ from Eq.~(27). 
    The true stack is recovered as $\mathbf{s}_t \defeq \mathbf{h}_t^{(\text{p})} + \mathbf{h}_t^{(\text{q})}$.

    Let $\mathbf{U}_{\text{push}}, \mathbf{U}_{\text{pop}} \in \mathbb{R}^{km\times km}$ be the shift matrices from part~(d) and let $\mathbf{V}_{\text{ins}} \in \mathbb{R}^{km \times k}$ be the matrix that places $\mathbf{e}_i$ into block~1 when its input is $\openBr{i}$, i.e.\ $\mathbf{V}_{\text{ins}} = (\mathbf{I}_k, \mathbf{0}_k, \dots, \mathbf{0}_k)^\top$ (the first $k$ columns of the $\mathbf{V}$ from part~(d)).
    Define
    \[
        \mathbf{U} \;=\; \begin{pmatrix} \mathbf{U}_{\text{push}} & \mathbf{U}_{\text{push}} \\ \mathbf{U}_{\text{pop}} & \mathbf{U}_{\text{pop}} \end{pmatrix} \in \mathbb{R}^{2km \times 2km},
        \qquad
        \mathbf{b} \;=\; -\tfrac{1}{2}\,\mathbf{1}_{2km} \in \mathbb{R}^{2km},
    \]
    \[
        \mathbf{V} \;=\;
        \begin{pmatrix}
            \mathbf{V}_{\text{ins}} & -2\,\mathbf{1}_{km \times k} & -2\,\mathbf{1}_{km \times 1} \\[2pt]
            -2\,\mathbf{1}_{km \times k} & \mathbf{0}_{km \times k} & -2\,\mathbf{1}_{km \times 1}
        \end{pmatrix} \in \mathbb{R}^{2km \times (2k+1)},
    \]
    where the $2k+1$ columns are ordered as $\openBr{1}, \dots, \openBr{k}, \closeBr{1}, \dots, \closeBr{k}, \eos$.

    \paragraph{Why $\mathbf{U}$ is correct.} Multiplying out,
    \[
        \mathbf{U}\mathbf{h}_{t-1} \;=\; \begin{pmatrix} \mathbf{U}_{\text{push}}(\mathbf{h}_{t-1}^{(\text{p})} + \mathbf{h}_{t-1}^{(\text{q})}) \\ \mathbf{U}_{\text{pop}}(\mathbf{h}_{t-1}^{(\text{p})} + \mathbf{h}_{t-1}^{(\text{q})}) \end{pmatrix} \;=\; \begin{pmatrix} \mathbf{U}_{\text{push}}\mathbf{s}_{t-1} \\ \mathbf{U}_{\text{pop}}\mathbf{s}_{t-1} \end{pmatrix},
    \]
    so we recover both candidate next-stacks from the true stack $\mathbf{s}_{t-1}$. 
    One of $\mathbf{h}_{t-1}^{(\text{p})}, \mathbf{h}_{t-1}^{(\text{q})}$ is $\mathbf{0}$ and the other is a $0/1$ encoding, hence $\mathbf{s}_{t-1} \in \{0, 1\}^{km}$ and the entries of $\mathbf{U}\mathbf{h}_{t-1}$ lie in $\{0, 1\}$.

    \paragraph{Correspondence with $\mathcal{A}$.} Suppose $\mathbf{s}_{t-1} = h(q)$ for some non-rejecting state $q = [\openBr{i_1}\cdots\openBr{i_{m'}}] \in \states$.
    
    
    \emph{Valid push} ($y_t = \openBr{i}$, $m' < m$, $q' = [\openBr{i_1}\cdots\openBr{i_{m'}}\openBr{i}]$). 
    The one-hot $\mathbf{o}_{y_t}$ selects column $i$ of $\mathbf{V}$:
    \[
        \mathbf{V}\mathbf{o}_{\openBr{i}} \;=\; \begin{pmatrix} \mathbf{V}_{\text{ins}}\,\mathbf{e}_i \\ -2\,\mathbf{1}_{km} \end{pmatrix} \;=\; \begin{pmatrix} (\mathbf{e}_i, \mathbf{0}_{(m-1)k})^\top \\ -2\,\mathbf{1}_{km} \end{pmatrix} \;\Rightarrow\; \mathbf{z}_t \;=\; \begin{pmatrix} \mathbf{U}_{\text{push}}\mathbf{s}_{t-1} + (\mathbf{e}_i, \mathbf{0})^\top - \tfrac{1}{2}\mathbf{1} \\[2pt] \mathbf{U}_{\text{pop}}\mathbf{s}_{t-1} - \tfrac{5}{2}\mathbf{1} \end{pmatrix}.
    \]
    The top half: $\mathbf{U}_{\text{push}}\mathbf{s}_{t-1}$ has its block~1 zeroed out, so adding $(\mathbf{e}_i, \mathbf{0})^\top$ never causes a collision.
    Every entry is $0$ or $1$, and after subtracting $\tfrac{1}{2}$ they sit at $\pm\tfrac{1}{2}$. 
    The Heaviside yields $h(q')$. 
    The bottom half: each entry is $\le 1 - \tfrac{5}{2} = -\tfrac{3}{2} < 0$, so $H$ makes it $\mathbf{0}$. 
    Thus $\mathbf{h}_t^{(\text{p})} = h(q')$, $\mathbf{h}_t^{(\text{q})} = \mathbf{0}$, and $\mathbf{s}_t = h(q')$.
    
    
    \emph{Valid pop} ($y_t = \closeBr{i}$, $i = i_{m'}$, $q' = [\openBr{i_1}\cdots\openBr{i_{m'-1}}]$). 
    The one-hot picks column $k+i$:
    \[
        \mathbf{V}\mathbf{o}_{\closeBr{i}} \;=\; \begin{pmatrix} -2\,\mathbf{1}_{km} \\ \mathbf{0}_{km} \end{pmatrix} \;\Rightarrow\; \mathbf{z}_t \;=\; \begin{pmatrix} \mathbf{U}_{\text{push}}\mathbf{s}_{t-1} - \tfrac{5}{2}\mathbf{1} \\[2pt] \mathbf{U}_{\text{pop}}\mathbf{s}_{t-1} - \tfrac{1}{2}\mathbf{1} \end{pmatrix}.
    \]
    The top half is $\le -\tfrac{3}{2} < 0$, so $H$ zeros it. 
    The bottom half has entries in $\{-\tfrac{1}{2}, +\tfrac{1}{2}\}$ and $H$ recovers $\mathbf{U}_{\text{pop}}\mathbf{s}_{t-1} = h(q')$. 
    Hence $\mathbf{h}_t^{(\text{p})} = \mathbf{0}$, $\mathbf{h}_t^{(\text{q})} = h(q')$, and $\mathbf{s}_t = h(q')$.
    
    
    \emph{$\eos$.} The last column of $\mathbf{V}$ is $-2\,\mathbf{1}_{2km}$, so both halves of $\mathbf{z}_t$ are $\le -\tfrac{3}{2}\mathbf{1}$ and $\mathbf{h}_t = \mathbf{0}_{2km}$. 
    Since the recognition module of part~(f) reads $\mathbf{h}_{t-1}$ to compute the probability of $y_t = \eos$, this loss of information at $t = T$ does not affect recognition.


    \paragraph{Invalid transitions.} A push on a full stack ($m' = m$) shifts the bottom block of $\mathbf{s}_{t-1}$ off the top half (annihilated by $\mathbf{U}_{\text{push}}$), 
    so $\mathbf{h}_t^{(\text{p})}$ encodes a \emph{different} length-$m$ stack instead of the rejecting state of $\mathcal{A}$. 
    Similarly, a mismatched or empty pop produces a top half that does not equal $h(q')$ for the true continuation. 
    The dynamics of part~(e) cannot detect these failures by themselves. 
    This is the role of the output layer constructed in part~(f), which assigns probability~$<\eta$ to such invalid continuations and
    thus excludes them from the locally $\eta$-truncated support. 
    On every valid prefix, however, the recurrence $\mathbf{h}_t = H(\mathbf{U}\mathbf{h}_{t-1} + \mathbf{V}\mathbf{o}_{y_t} + \mathbf{b})$ tracks the transitions of $\mathcal{A}$ exactly via $\mathbf{s}_t$.
\end{subquestion}

\begin{subquestion}
    Recall that one of the two halves of $\mathbf{h}_{t-1}$ is $\mathbf{0}$ and the other equals the encoding $\mathbf{s}_{t-1} = h(q)$ of the current automaton state $q$ via Eq.~(27). 
    Thus $\mathbf{s}_{t-1}$ is composed of $m$ blocks of size $k$, with the top of the stack in block $j = 1$ and the bottom in block $j = m$ (zero-padded above the stack depth). 
    For brevity, we work with $\mathbf{s}_{t-1}$ as a stand-in for $\mathbf{h}_{t-1}$ below. 
    The same logits are obtained by feeding either half (the all-zero half contributes nothing). 
    Write $\mathbf{s}^{(j)} \in \mathbb{R}^{k}$ for the $j$-th block. 
    Three indicators are linear in $\mathbf{s}_{t-1}$ and take values in $\{0, 1\}$ on every valid encoding:
    \[
        \mathbb{I}_{\text{full}} = \mathbf{1}_k^\top \mathbf{s}^{(m)}, \qquad \mathbb{I}_{\text{empty}} = 1 - \mathbf{1}_k^\top \mathbf{s}^{(1)}, \qquad \mathbb{I}_{\text{top}=i} = \mathbf{s}^{(1)}_i.
    \]

    Design $\mathbf{E}\mathbf{s}_{t-1} + \mathbf{b}'$ so that valid continuations yield a logit of $0$ and invalid ones $-\alpha$. 
    Order the rows of $\mathbf{E}$ and $\mathbf{b}'$ as $\openBr{1}, \dots, \openBr{k}, \closeBr{1}, \dots, \closeBr{k}, \eos$. 
    Let $\mathbf{J}_k = \mathbf{1}_k\mathbf{1}_k^\top$ be the $k \times k$ ones matrix.
    \[
        \mathbf{E} \;=\;
        \begin{pmatrix}
            \mathbf{0}_{k\times k} & \mathbf{0}_{k\times k} & \cdots & \mathbf{0}_{k\times k} & -\alpha\,\mathbf{J}_k \\[2pt]
            \alpha\,\mathbf{I}_k   & \mathbf{0}_{k\times k} & \cdots & \mathbf{0}_{k\times k} & \mathbf{0}_{k\times k} \\[2pt]
            -\alpha\,\mathbf{1}_k^\top & \mathbf{0}_{1\times k} & \cdots & \mathbf{0}_{1\times k} & \mathbf{0}_{1\times k}
        \end{pmatrix} \in \mathbb{R}^{(2k+1) \times km},
        \qquad
        \mathbf{b}' \;=\;
        \begin{pmatrix} \mathbf{0}_{k} \\ -\alpha\,\mathbf{1}_{k} \\ 0 \end{pmatrix}.
    \]
    \begin{align*}
        z_{\openBr{i}} \;&=\; -\alpha\,\mathbf{1}_k^\top \mathbf{s}^{(m)} \;=\; -\alpha\,\mathbb{I}_{\text{full}}, \\
        z_{\closeBr{i}} \;&=\; \alpha\,\mathbf{s}^{(1)}_i - \alpha \;=\; -\alpha\,(1 - \mathbb{I}_{\text{top}=i}), \\
        z_{\eos} \;&=\; -\alpha\,\mathbf{1}_k^\top \mathbf{s}^{(1)} \;=\; -\alpha\,(1 - \mathbb{I}_{\text{empty}}).
    \end{align*}
    These are $0$ exactly when the corresponding symbol is an acceptable continuation under $\trans$ (push allowed when not full, pop allowed when its bracket matches the top, $\eos$ allowed when the stack is empty), and $-\alpha$ otherwise.

    Because the logit of \emph{every} valid continuation is $0$ and that of \emph{every} invalid one is $-\alpha$, the partition function $Z = \sum_{y \in \overline{\alphabet}} \exp(z_y)$ depends only on how many continuations are valid. There are exactly three cases.
    \begin{enumerate}
        \item \emph{Empty stack} ($m' = 0$): the $k$ pushes and $\eos$ are valid, the $k$ pops are invalid;
        $Z_{\text{empty}} = (k+1) + k e^{-\alpha}$.
        \item \emph{Partially full stack} ($0 < m' < m$): the $k$ pushes and the unique matching pop are valid; the $k-1$ mismatched pops and $\eos$ are invalid;
        $Z_{\text{partial}} = (k+1) + k e^{-\alpha}$.
        \item \emph{Full stack} ($m' = m$): only the matching pop is valid; the $k$ pushes, $k-1$ mismatched pops, and $\eos$ are all invalid;
        $Z_{\text{full}} = 1 + 2k e^{-\alpha}$.
    \end{enumerate}

    \paragraph{Choosing $\alpha$.} Let $N_{\text{val}} \in \{k+1, 1\}$ denote the number of valid continuations in the current case. 
    Valid and invalid probabilities are $1/Z$ and $e^{-\alpha}/Z$ respectively, and the locally $\eta$-truncated support condition $\mathcal{L}_{\eta}(p_{\text{SM}}) = D(k, m)$ requires
    \[
        \frac{1}{Z} \;\ge\; \eta \;\;\text{(valid)}, \qquad \frac{e^{-\alpha}}{Z} \;<\; \eta \;\;\text{(invalid)}.
    \]
    As $\alpha \to \infty$, $e^{-\alpha} \to 0$, so $Z \to N_{\text{val}}$, the valid probability tends to $1/N_{\text{val}} \in \{\tfrac{1}{k+1}, 1\}$, and the invalid probability tends to $0$. 
    Since the assignment guarantees $\eta < \tfrac{1}{k+1} \le 1/N_{\text{val}}$, the limit valid probability exceeds $\eta$ in every case, and the limit invalid probability is below $\eta$. 
    By continuity, there exists $\alpha^{*}(\eta) > 0$ such that both inequalities hold in all three cases simultaneously for every $\alpha \ge \alpha^{*}(\eta)$, so the construction goes through for sufficiently large $\alpha$ (cf.\ the footnote in the assignment).
    Two conditions translate into upper bounds on $e^{-\alpha}$:

    \emph{Valid condition} ($1/Z \ge \eta$, i.e.\ $Z \le 1/\eta$):
    \[
        e^{-\alpha} \;\le\; \tfrac{1/\eta - (k+1)}{k} \quad \text{(Cases 1, 2)},
        \qquad
        e^{-\alpha} \;\le\; \tfrac{1/\eta - 1}{2k} \quad \text{(Case 3)}.
    \]

    \emph{Invalid condition} ($e^{-\alpha}/Z < \eta$, i.e.\ $e^{-\alpha} < \eta\, Z$):
    \[
        e^{-\alpha} \;<\; \tfrac{\eta(k+1)}{1 - k\eta} \quad \text{(Cases 1, 2; \;$\eta < \tfrac{1}{k+1} < \tfrac{1}{k}$ ensures $1 - k\eta > 0$)},
    \]
    \[
        e^{-\alpha} \;<\; \tfrac{\eta}{1 - 2k\eta} \quad \text{(Case 3; automatic when $\eta \ge \tfrac{1}{2k}$)}.
    \]
    Taking $\alpha$ at least the maximum of the four logarithmic thresholds with strict inequality where required suffices to guarantee $\mathcal{L}_{\eta}(p_{\text{SM}}) = D(k, m)$.
\end{subquestion}
